```
React Web应用 → PWA → Electron桌面版 → 移动端适配
```

## 详细技术栈

### 前端技术栈
- **核心框架**：React 18 + TypeScript
- **UI框架**：Tailwind CSS + Headless UI
- **状态管理**：Zustand 或 Redux Toolkit
- **路由**：React Router
- **PWA**：Workbox (离线缓存、推送通知)

### 后端技术栈
- **服务器**：Node.js + Express 或 Python + FastAPI
- **数据库**：PostgreSQL (主库) + Redis (缓存)
- **文件存储**：AWS S3 或 阿里云OSS
- **搜索引擎**：Elasticsearch
- **实时通信**：Socket.io (论坛实时聊天)

### 跨平台部署
- **Web版**：Vercel/Netlify 部署
- **桌面版**：Electron + Electron Builder
- **移动版**：PWA安装 + 后期考虑React Native

```typescript
import { pool } from '../config/database';
import bcrypt from 'bcryptjs';

export interface User {
  id: number;
  username: string;
  email: string;
  avatar_url?: string;
  role: 'reader' | 'author' | 'admin';
  vip_level: number;
  coins: number;
  created_at: Date;
}

export class UserModel {
  static async create(userData: {
    username: string;
    email: string;
    password: string;
    role?: string;
  }): Promise<User> {
    const hashedPassword = await bcrypt.hash(userData.password, 12);
    
    const query = `
      INSERT INTO users (username, email, password_hash, role)
      VALUES ($1, $2, $3, $4)
      RETURNING id, username, email, avatar_url, role, vip_level, coins, created_at
    `;
    
    const result = await pool.query(query, [
      userData.username,
      userData.email,
      hashedPassword,
      userData.role || 'reader'
    ]);
    
    return result.rows[0];
  }

  static async findByEmail(email: string): Promise<User | null> {
    const query = `
      SELECT id, username, email, password_hash, avatar_url, role, vip_level, coins, created_at
      FROM users WHERE email = $1
    `;
    const result = await pool.query(query, [email]);
    return result.rows[0] || null;
  }

  static async findById(id: number): Promise<User | null> {
    const query = `
      SELECT id, username, email, avatar_url, role, vip_level, coins, created_at
      FROM users WHERE id = $1
    `;
    const result = await pool.query(query, [id]);
    return result.rows[0] || null;
  }

  static async updateProfile(id: number, data: Partial<User>): Promise<User> {
    const fields = [];
    const values = [];
    let paramCount = 1;

    for (const [key, value] of Object.entries(data)) {
      if (value !== undefined && key !== 'id') {
        fields.push(`${key} = $${paramCount}`);
        values.push(value);
        paramCount++;
      }
    }

    values.push(id);
    
    const query = `
      UPDATE users SET ${fields.join(', ')}, updated_at = NOW()
      WHERE id = $${paramCount}
      RETURNING id, username, email, avatar_url, role, vip_level, coins, created_at
    `;
    
    const result = await pool.query(query, values);
    return result.rows[0];
  }
}

import { Request, Response } from 'express';
import jwt from 'jsonwebtoken';
import bcrypt from 'bcryptjs';
import { UserModel } from '../models/User';
import { validationResult } from 'express-validator';

export class AuthController {
  static async register(req: Request, res: Response) {
    try {
      const errors = validationResult(req);
      if (!errors.isEmpty()) {
        return res.status(400).json({
          success: false,
          message: '输入数据验证失败',
          errors: errors.array()
        });
      }

      const { username, email, password, role } = req.body;

      // 检查用户是否已存在
      const existingUser = await UserModel.findByEmail(email);
      if (existingUser) {
        return res.status(400).json({
          success: false,
          message: '该邮箱已被注册'
        });
      }

      // 创建新用户
      const user = await UserModel.create({ username, email, password, role });

      // 生成JWT token
      const token = jwt.sign(
        { 
          userId: user.id, 
          username: user.username, 
          role: user.role,
          vipLevel: user.vip_level
        },
        process.env.JWT_SECRET!,
        { expiresIn: process.env.JWT_EXPIRES_IN || '7d' }
      );

      res.status(201).json({
        success: true,
        message: '注册成功',
        data: {
          token,
          user: {
            id: user.id,
            username: user.username,
            email: user.email,
            role: user.role,
            vip_level: user.vip_level,
            coins: user.coins
          }
        }
      });
    } catch (error) {
      console.error('注册错误:', error);
      res.status(500).json({
        success: false,
        message: '服务器内部错误'
      });
    }
  }

  static async login(req: Request, res: Response) {
    try {
      const errors = validationResult(req);
      if (!errors.isEmpty()) {
        return res.status(400).json({
          success: false,
          message: '输入数据验证失败',
          errors: errors.array()
        });
      }

      const { email, password } = req.body;

      // 查找用户
      const user = await UserModel.findByEmail(email);
      if (!user) {
        return res.status(400).json({
          success: false,
          message: '邮箱或密码错误'
        });
      }

      // 验证密码
      const isPasswordValid = await bcrypt.compare(password, (user as any).password_hash);
      if (!isPasswordValid) {
        return res.status(400).json({
          success: false,
          message: '邮箱或密码错误'
        });
      }

      // 生成JWT token
      const token = jwt.sign(
        { 
          userId: user.id, 
          username: user.username, 
          role: user.role,
          vipLevel: user.vip_level
        },
        process.env.JWT_SECRET!,
        { expiresIn: process.env.JWT_EXPIRES_IN || '7d' }
      );

      res.json({
        success: true,
        message: '登录成功',
        data: {
          token,
          user: {
            id: user.id,
            username: user.username,
            email: user.email,
            role: user.role,
            vip_level: user.vip_level,
            coins: user.coins
          }
        }
      });
    } catch (error) {
      console.error('登录错误:', error);
      res.status(500).json({
        success: false,
        message: '服务器内部错误'
      });
    }
  }

  static async getProfile(req: Request, res: Response) {
    try {
      const userId = (req as any).user.userId;
      const user = await UserModel.findById(userId);
      
      if (!user) {
        return res.status(404).json({
          success: false,
          message: '用户不存在'
        });
      }

      res.json({
        success: true,
        data: {
          user: {
            id: user.id,
            username: user.username,
            email: user.email,
            avatar_url: user.avatar_url,
            role: user.role,
            vip_level: user.vip_level,
            coins: user.coins,
            created_at: user.created_at
          }
        }
      });
    } catch (error) {
      console.error('获取用户信息错误:', error);
      res.status(500).json({
        success: false,
        message: '服务器内部错误'
      });
    }
  }
}
```
